class IntelligentLectureSummarizer:
    def __init__(self):
        self.models = {
            "math": self.load_model("math-specialist-bart"),
            #"physics": self.load_model("physics-flan-t5"),
            "cs": self.load_model("cs-llama-adapter"),
            "general": self.load_model("general-bart")
        }
        
        # Use your 02_train_stem_ner for subject detection
        self.subject_detector = STEMSubjectClassifier()
    
    def summarize(self, transcript, frames_metadata):
        # Step 1: Detect subject using NER patterns
        subject = self.subject_detector.predict(transcript)
        
        # Step 2: Route to appropriate model
        if subject in self.models:
            base_summary = self.models[subject].generate(transcript)
        else:
            base_summary = self.models["general"].generate(transcript)
        
        # Step 3: Enhance with LLM if needed (configurable)
        if self.use_enhancement:
            enhanced = self.llm_enhance(base_summary, transcript, frames_metadata)
            return enhanced
        
        return base_summary
    
    def llm_enhance(self, draft_summary, context, visual_metadata):
        # Use larger LLM for refinement
        prompt = f"""
        As an expert teaching assistant, enhance this lecture summary.
        
        CONTEXT: {context[:1500]}
        
        DRAFT SUMMARY: {draft_summary}
        
        VISUAL ELEMENTS: {visual_metadata}
        
        Requirements:
        1. Ensure all key equations/concepts from context are included
        2. Reference visual elements appropriately
        3. Structure for student revision (bullet points if helpful)
        4. Keep under 150 words
        
        ENHANCED SUMMARY:
        """
        return self.llm_api.generate(prompt)"
